# База анти-сленга для линта отчётов (механизм 5).
# Формат строк: категория<TAB>regex (Python re, компиляция с флагом IGNORECASE).
# Все категории — FAIL. Комментарии — строки, начинающиеся с #.

# От первого лица — в отчёте допустима только безличная констатация.
first_person	\b(я|мы|нас|нам|наш|наши|вы|вас|вам|ваш)\b

# Фразы агента-собеседника (чат-мусор), недопустимые в тексте отчёта.
agent_phrases	\b(давайте|рассмотрим|я проанализировал|на мой взгляд|к сожалению|итак|по сути|в целом|готово|отлично|супер)\b

# Маркетинговые оценки без фактов.
marketing	\b(инновационн\w+|уникальн\w+|лидер рынка|лучшее решение|революционн\w+)\b

# Офисные англицизмы и жаргон.
office_anglicisms	\b(деливери|фидбек|синк|капля|хайлайт\w*|чекнуть|запушить|мерж\w*|реюз\w*|болванка|скелет отчёта|апдейтнуть)\b

# Эмодзи и пиктограммы (доп. символы, стрелки-декор, dingbats).
emoji	[\U0001F300-\U0001FAFF\u2600-\u27BF]

# Остатки markdown-разметки: **жирный**, заголовки #, бэктики, разделители таблиц |---.
markdown_residue	(\*\*|^#{1,6}\s|\x60|\|---)

# Восклицательные знаки — отчёт констатирует, а не восклицает.
exclamations	!

# Незаполненные плейсхолдеры — прямой признак несформированного отчёта.
placeholders	(\$unresolved|\bTBD\b|\bTODO\b|\bFIXME\b|\?\?\?)
